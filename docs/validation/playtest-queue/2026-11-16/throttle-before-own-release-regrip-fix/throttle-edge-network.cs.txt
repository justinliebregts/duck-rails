// Install this SAME command once on each of exactly two already-running authenticated clients.
// Change RUN on BOTH clients for a rerun; stale rendezvous files are never reused.
// <=85s including rendezvous, ~34s after synchronized start; no distance/room-property seeding.
// 1. Concurrent requests + losing heartbeat/release: controlled public network API calls.
// 2. Acquire/release before observable grant: same-callback requests on non-master.
// 3. Ordinary simulated right-hand grip/arc, real peer pose stream, regrip during return,
//    then controlled OnApplicationFocus(false) callback. No claim of OS focus or real hands.
// Regrip deliberately extends driver-local returnDuration to2.5s; restored afterward.
// This proves preservation of the current angle across a live regrip, NOT default0.25s reach.
// Actual menu input interruption is NOT covered. Begin stopped, awake, empty-handed, menu closed.
// Run alone: no other fixture, human input or scene transitions. Inspect BOTH output END lines.

internal class CommandScript:IRunCommand {
 public void Execute(ExecutionResult r){
  if(!Application.isPlaying||!PhotonNetwork.InRoom||PhotonNetwork.CurrentRoom.PlayerCount!=2||PhotonSession.Phase!=RailPhase.Running||!PhotonSession.Instance.WorldReady)
   throw new Exception("Requires exactly two ready authenticated clients in Running");
  if(GameObject.Find("Bounded throttle network edges")!=null)throw new Exception("Edge fixture already installed");
  var g=new GameObject("Bounded throttle network edges");var probe=g.AddComponent<QueueThrottleEdges>();g.AddComponent<QueueThrottleLate>().owner=probe;r.RegisterObjectCreation(g);r.Log("Armed throttle edges; install same RUN on peer, then read both END records");
 }
}

   Check(PhotonSession.DriveGripActor==0,"pre-ack case starts unowned");
   PhotonSession.Instance.RequestDriveGrip(run,token+"-early",0,0);
   PhotonSession.Instance.RequestDriveGrip(run,token+"-early",2,0);
   Check(PhotonSession.DriveGripActor==0,"non-master queued acquire/release before any observable grant in same callback");}
  if(age>12&&Once("earlyCheck"))Check(PhotonSession.DriveGripActor==0&&Mathf.Abs(Target)<.001f,"pre-ack release leaves no ghost lease on this client");

  // Informational old-phase measurement only: train has moved, remote LateUpdate has not.
  if(age>19&&age<21&&actor!=driver&&!peerChecked&&File.Exists(HandFile)){
   var sample=JsonUtility.FromJson<HandSample>(File.ReadAllText(HandFile));var avatar=PhotonPlayer.ForActor(driver);
   if(avatar!=null&&avatar.TryGetHandPose(false,out var point,out _)){preLateFrame=Time.frameCount;preLateGap=Vector3.Distance(RailTrain.Instance.transform.InverseTransformPoint(point),sample.local);}}
  if(age>30&&Once("focusCheck"))Check(PhotonSession.DriveGripActor==0&&Mathf.Abs(Target)<.001f,"controlled focus-loss callback settles shared lease/target to stopped");
  if(age>34){Check(once.Contains("contestCheck")&&once.Contains("earlyCheck")&&once.Contains("focusCheck")&&(actor==driver?once.Contains("earlyRelease")&&once.Contains("physicalCheck")&&once.Contains("regripCheck")&&once.Contains("detachCheck"):peerChecked),"all required phases actually sampled");Check(PhotonSession.DriveGripActor==0&&Mathf.Abs(Target)<.001f,"final shared state stopped/unowned");Finish();}
 }catch(Exception e){Fail(e);}}

   nextHandSample=PhotonNetwork.Time+.1;

   regripStarted=true;regripExpected=travel;Log("INPUT fresh actual grip during visible return progress="+travel);}
  if(regripStarted&&age<27){grip=true;travel=Get<float>(drive,"gripStartProgress");if(!regripHeld)travel=regripExpected;}
  if(age>=27){grip=false;if(Once("focusInput")){RailRemoteBridge.Call(drive,"OnApplicationFocus",false);Log("INPUT controlled OnApplicationFocus(false) callback; OS focus not manipulated");}}
  var target=drive.transform.TransformPoint(Quaternion.Euler(travel*70,0,0)*offPalm);
  Vector3 offset=Palm.position-body.RightHandAnchor.position;
  sim.InjectedInput=new SimulatedRigInput.Frame{grip=grip,rightReach=true};
  RailRemoteBridge.Set(sim,"rightPosition",body.RightHandAnchor.parent.InverseTransformPoint(target-offset));RailRemoteBridge.Call(sim,"WriteSimulatedPose");
  if(age>18&&age<21){
   if(!firstHeld){Check(drive.HeldBy(Hand)&&PhotonSession.DriveGripActor==actor,"ordinary physical grip acquires non-master lease");firstToken=Get<string>(drive,"gripToken");firstHeld=true;}

  }

  if(age>29&&Once("detachCheck"))Check(!drive.HeldBy(Hand)&&!RailControl.TryGetDrivePalm(Palm,out _,out _),"focus-loss callback releases physical grip and rendered palm constraint");
 }

   if(!oldKinematic){motor.rb.linearVelocity=oldVelocity;motor.rb.angularVelocity=oldAngularVelocity;}
   Log("RESTORED rig injection, stored poses, movement/rigidbody, returnDuration, original train-relative position; no train distance reset");
  }
