  actor=PhotonNetwork.LocalPlayer.ActorNumber;room=PhotonNetwork.CurrentRoom.Name;run=PhotonSession.RunId;
  string root=Application.dataPath.Replace('\\','/');int vp=root.IndexOf("/Library/VP/",StringComparison.OrdinalIgnoreCase);root=vp<0?Path.GetDirectoryName(root):root.Substring(0,vp);path=root+"/docs/validation/playtest-queue/2026-11-16/throttle-narrow-postfix-actor"+actor+".txt";

  origin=RailTrain.Instance.transform.InverseTransformPoint(new Vector3(motor.headCollider.transform.position.x,motor.transform.position.y,motor.headCollider.transform.position.z));heading=Quaternion.Inverse(RailTrain.Instance.transform.rotation)*motor.transform.rotation;
  foreach(string n in new[]{"headPosition","leftPosition","rightPosition","headRotation","leftRotation","rightRotation","yaw","pitch","reach","handRoll","awaitNeutral","awaitThrottleNeutral","throttleKeyWasHeld","Buttons","ThrottleHeld"})poses[n]=RailRemoteBridge.Get(sim,n);

}
