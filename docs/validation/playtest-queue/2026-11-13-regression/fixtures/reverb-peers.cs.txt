using System;using System.IO;using System.Linq;using UnityEditor;using UnityEngine;using Photon.Pun;using Game.Audio;
internal class CommandScript:IRunCommand{string role,path,room;BoxReverbZone target;Vector3 origin;bool kin;double at;int stage;const string Root="X:/Work/duck-rails/docs/validation/playtest-queue/2026-11-13-regression/";void Log(string s)=>File.AppendAllText(path,DateTime.UtcNow.ToString("O")+" "+s+"\n");
public void Execute(ExecutionResult r){role=Unity.Multiplayer.PlayMode.CurrentPlayer.Tags.First();path=Root+"reverb-independent-"+role+".txt";room=PhotonNetwork.CurrentRoom.Name;target=UnityEngine.Object.FindObjectsByType<BoxReverbZone>(FindObjectsSortMode.None).First(x=>x.name==(role=="Player2"?"Arcade":"West bunker"));origin=NewPlayer.instance.transform.position;kin=NewPlayer.instance.rb.isKinematic;NewPlayer.instance.rb.isKinematic=true;at=EditorApplication.timeSinceStartup;EditorApplication.update+=Tick;r.Log("Independent listener "+role+" zone="+target.name);}
void Tick(){try{var s=PhotonSession.Instance;if(s==null||!s.WorldReady||s.HasRoomRequest)return;double age=EditorApplication.timeSinceStartup-at;if(age>90){Log("BLOCKED timeout stage="+stage);EditorApplication.update-=Tick;return;}
if(stage==0){var listener=AudioManager.instance.ListenerTransform;NewPlayer.instance.transform.position+=target.transform.position-listener.position;NewPlayer.instance.rb.position=NewPlayer.instance.transform.position;NewPlayer.instance.ReseedTrackingPose();if(age<2)return;stage=1;at=EditorApplication.timeSinceStartup;}
else if(stage==1&&age>2){var zone=(AudioReverbZone)RailRemoteBridge.Get(target,"zone");var listener=AudioManager.instance.ListenerTransform;var other=UnityEngine.Object.FindObjectsByType<BoxReverbZone>(FindObjectsSortMode.None).First(x=>x.name==(role=="Player2"?"West bunker":"Arcade"));var oz=(AudioReverbZone)RailRemoteBridge.Get(other,"zone");Log((Vector3.Distance(zone.transform.position,listener.position)<.02f&&Vector3.Distance(oz.transform.position,listener.position)>oz.maxDistance?"PASS ":"FAIL ")+"local listener inside "+target.name+" only; other client region outside; targetDistance="+Vector3.Distance(zone.transform.position,listener.position));NewPlayer.instance.rb.isKinematic=kin;NewPlayer.instance.Warp(origin);if(role=="Player3"){s.RequestJoinRandom();stage=2;at=EditorApplication.timeSinceStartup;}else{Log("DONE");EditorApplication.update-=Tick;}}
else if(stage==2&&PhotonSession.Phase==RailPhase.Compound){Log((UnityEngine.Object.FindObjectsByType<BoxReverbZone>(FindObjectsSortMode.None).Length==0?"PASS ":"FAIL ")+"Expedition unload removes block reverb components; remaining="+UnityEngine.Object.FindObjectsByType<BoxReverbZone>(FindObjectsSortMode.None).Length);s.RequestJoinCode(room);stage=3;at=EditorApplication.timeSinceStartup;}
else if(stage==3&&PhotonSession.Phase==RailPhase.Running){Log((UnityEngine.Object.FindObjectsByType<BoxReverbZone>(FindObjectsSortMode.None).Length>0?"PASS ":"FAIL ")+"block reverb recreated on rejoin");Log("DONE");EditorApplication.update-=Tick;}}
catch(Exception e){Log("FIXTURE ERROR "+e);EditorApplication.update-=Tick;}}
}
