using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using Photon.Pun;
internal class CommandScript:IRunCommand{const string Root="X:/Work/duck-rails/";string role,path,room;int stage,warmFrames,uncovered;double at,joined;bool first,kin,replay;int lastSeq;void Log(string s)=>File.AppendAllText(path,DateTime.UtcNow.ToString("O")+" "+s+"\n");
public void Execute(ExecutionResult r){role=Unity.Multiplayer.PlayMode.CurrentPlayer.IsMainEditor?"main":Unity.Multiplayer.PlayMode.CurrentPlayer.Tags.First();path=Root+"docs/validation/playtest-queue/2026-11-13-regression/revive-late-"+role+".txt";File.WriteAllText(path,"Real late join; seeded down status, actual helper proximity and authoritative progress/completion.\n");room=PhotonNetwork.CurrentRoom.Name;at=EditorApplication.timeSinceStartup;if(role=="main"){File.WriteAllText(Root+"Temp/revive-late-room.txt",room);File.WriteAllText(Root+"Temp/revive-late-join.txt","");File.WriteAllText(Root+"Temp/revive-late-observed.txt","");PhotonNetwork.SetMasterClient(PhotonNetwork.LocalPlayer);}if(role=="Player3")PhotonSession.Instance.LeaveCurrentRoom();EditorApplication.update+=Tick;}
void Place(Vector3 local){var p=NewPlayer.instance;if(!first){first=true;kin=p.rb.isKinematic;}p.rb.isKinematic=true;p.transform.position+=RailTrain.Instance.transform.TransformPoint(local)-p.headCollider.transform.position;p.rb.position=p.transform.position;p.ReseedTrackingPose();}
void Tick(){try{double now=EditorApplication.timeSinceStartup;if(now-at>110){Log("BLOCKED timeout stage="+stage);End();return;}foreach(var h in UnityEngine.Object.FindObjectsByType<PhotonMasterHandoff>(FindObjectsSortMode.None))RailRemoteBridge.Set(h,"focused",true);
if(role=="Player3"){ if((stage==1||stage==4)&&ExpeditionRenderWarmup.Running){warmFrames++;if(!RailTransitionVeil.Covered)uncovered++;}
if(stage==0&&PhotonSession.Phase==RailPhase.Compound&&PhotonSession.Instance.WorldReady){var gate=UnityEngine.Object.FindFirstObjectByType<RailLaunchButton>();Log((gate!=null&&!gate.button.gameObject.activeSelf&&!gate.prompt.gameObject.activeSelf&&gate.prompt.font.name.Contains("CutePixel")?"PASS ":"FAIL ")+"public gate hidden; CutePixel prompt retained");File.WriteAllText(Root+"Temp/revive-late-join.txt",DateTime.UtcNow.ToString("O"));PhotonSession.Instance.RequestJoinCode(room);stage=1;}
else if(stage==1&&RailExpedition.Instance!=null&&PhotonSession.Phase==RailPhase.Running){var run=RailExpedition.Instance;var target=run.State.players.FirstOrDefault(p=>p.name=="lebo");if(target==null)return;Place(new Vector3(-1,1.8f,2));var root=UnityEngine.Object.FindObjectsByType<Transform>(FindObjectsSortMode.None).FirstOrDefault(x=>x.name=="Revive feedback");var audio=root==null?null:root.GetComponents<AudioSource>().FirstOrDefault(x=>x.loop&&x.isPlaying);if(audio!=null&&target.down&&target.reviveAt>0){float expected=.25f+RailReviveEffects.Progress(target,PhotonNetwork.Time);Log((Mathf.Abs(audio.pitch-expected)<.08f?"PASS ":"FAIL ")+"late progress sound pitch="+audio.pitch+" expected="+expected+" progressAge="+(PhotonNetwork.Time-target.reviveAt)+" worldReady="+PhotonSession.Instance.WorldReady);File.WriteAllText(Root+"Temp/revive-late-observed.txt","seen");stage=2;lastSeq=target.reviveSequence;}}
else if(stage==2&&RailExpedition.Instance.State.players.First(p=>p.name=="lebo").reviveSequence>lastSeq){Log("Completion received after late progress join");PhotonSession.Instance.LeaveCurrentRoom();stage=3;}
else if(stage==3&&PhotonSession.Phase==RailPhase.Compound&&PhotonSession.Instance.WorldReady){bool clean=!UnityEngine.Object.FindObjectsByType<Transform>(FindObjectsSortMode.None).Any(x=>x.name=="Revive feedback");Log((clean?"PASS ":"FAIL ")+"Compound unload removes feedback roots");PhotonSession.Instance.RequestJoinCode(room);stage=4;}
else if(stage==4&&PhotonSession.Phase==RailPhase.Running&&PhotonSession.Instance.WorldReady){Log((uncovered==0&&!ExpeditionRenderWarmup.LastTimedOut?"PASS ":"FAIL ")+"Compound late join warmup coverage frames="+warmFrames+" uncovered="+uncovered+" newStates="+ExpeditionRenderWarmup.LastNewStates);joined=now;stage=5;}
else if(stage==5){var sounds=UnityEngine.Object.FindObjectsByType<Transform>(FindObjectsSortMode.None).Where(x=>x.name=="Revive feedback").SelectMany(x=>x.GetComponents<AudioSource>()).ToArray();if(sounds.Any(x=>!x.loop&&x.isPlaying))replay=true;if(now-joined>4){Log((replay?"FAIL ":"PASS ")+"post-completion late join sampled four seconds; historical replay="+replay);End();}}return;}
if(PhotonSession.Phase!=RailPhase.Running||RailExpedition.Instance==null)return;Place(role=="main"?new Vector3(0,1.8f,-2):new Vector3(.45f,1.8f,-2));
if(role=="main"&&PhotonNetwork.IsMasterClient&&File.ReadAllText(Root+"Temp/revive-late-join.txt").Length>0){var run=RailExpedition.Instance;var target=run.State.players.First(p=>p.name=="lebo");bool observed=File.ReadAllText(Root+"Temp/revive-late-observed.txt").Length>0;if(!observed&&!target.down){var state=JsonUtility.FromJson<RailRunState>(JsonUtility.ToJson(run.State));var p=state.players.First(x=>x.name=="lebo");p.down=true;p.reviveAt=0;p.reviveLostAt=0;p.rescuer="";RailRemoteBridge.Commit(RailExpedition.StateKey,JsonUtility.ToJson(state));}if(observed&&!target.down){Log("PASS authoritative rescue after joining in progress");End();}}
if(role=="Player2"&&File.ReadAllText(Root+"Temp/revive-late-observed.txt").Length>0&&!RailExpedition.Instance.State.players.First(p=>p.name=="lebo").down)End();
}catch(Exception e){Log("FIXTURE ERROR "+e);End();}}
void End(){if(first&&NewPlayer.instance!=null)NewPlayer.instance.rb.isKinematic=kin;EditorApplication.update-=Tick;Log("DONE");}
}
