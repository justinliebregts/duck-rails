using System;using System.IO;using System.Linq;using System.Collections.Generic;using System.Globalization;
using UnityEngine;using Photon.Pun;using Game.Audio;
internal class CommandScript:IRunCommand{
 public void Execute(ExecutionResult r){
  if(!Application.isPlaying||PhotonSession.Phase!=RailPhase.Compound||PhotonSession.IsPrivate||
   PhotonSession.Instance==null||!PhotonSession.Instance.WorldReady||PhotonSession.Instance.HasRoomRequest||
   PhotonNetwork.PlayerList.Count(p=>!p.IsInactive)!=2)throw new Exception("Need two ready clients in the same public Compound");
  if(PhotonSession.Get(RailSessionRules.Deadline,0d)>0)throw new Exception("Start from idle Compound with no countdown");
  if(NewPlayer.instance==null||SimulatedRigInput.Instance==null||!SimulatedRigInput.Instance.IsSimulating||SimulatedRigInput.Instance.UseInjectedInput)
   throw new Exception("Need Keyboard shared rig without another input fixture");
  var g=new GameObject("Two-client controlled Compound occupancy");g.AddComponent<CompoundMultiQueue>();
  UnityEngine.Object.DontDestroyOnLoad(g);r.RegisterObjectCreation(g);r.Log("Compound multiplayer occupancy armed; inject peer first, main creates shared start at Photon.Time + 3 seconds");
 }
}
[DefaultExecutionOrder(30000)] internal class CompoundMultiQueue:MonoBehaviour{
 const string Root="X:/Work/duck-rails/",Ev=Root+"docs/validation/playtest-queue/2026-11-16/",Sync=Root+"Temp/queue-compound-multi-start.txt";
 string path,room,role;bool main,synced,inside,pressed,done,firstSeen,bothSeen,oneSeen,cancelSeen,restartSeen,departed;
 double armed,start,syncReal,syncAge,firstDeadline,secondDeadline,nextLog;int mainActor,peerActor,starts,stops,ons,offs,downSamples,upSamples;
 RailPressurePlate plate;SoundEvent on,off;Vector3 rest,center,size,inPoint,outPoint,originalStand;Quaternion rotation;
 float maxDrop,moveError,sizeError,rotationError;double edge;
 readonly HashSet<string> seenAudio=new HashSet<string>(),failures=new HashSet<string>();
 void Log(string s)=>File.AppendAllText(path,DateTime.UtcNow.ToString("O")+" "+s+"\n");
 void Check(bool ok,string message){if(!ok&&failures.Add(message))Log("FAIL "+message);}
 void Start(){
  main=Unity.Multiplayer.PlayMode.CurrentPlayer.IsMainEditor;role=main?"main":Unity.Multiplayer.PlayMode.CurrentPlayer.Tags.FirstOrDefault()??"peer";
  room=PhotonNetwork.CurrentRoom.Name;path=Ev+"compound-multi-"+role+".txt";armed=Time.realtimeSinceStartupAsDouble;
  File.WriteAllText(path,"Unity="+Application.unityVersion+" clients=2 room="+room+" actor="+PhotonNetwork.LocalPlayer.ActorNumber+"\nControlled Warp occupancy, not natural locomotion. Audio evidence counts fresh leases, not audible quality. Master is not assumed.\n");
  plate=UnityEngine.Object.FindFirstObjectByType<RailPressurePlate>();
  if(plate==null||plate.plate==null||plate.contactSurface==null){Log("BLOCKED pressure plate references missing");done=true;return;}
  rest=(Vector3)RailRemoteBridge.Get(plate,"rest");center=plate.contactSurface.bounds.center;size=plate.contactSurface.bounds.size;rotation=plate.contactSurface.transform.rotation;
  on=plate.pressSound;off=plate.releaseSound;pressed=plate.Pressed;edge=armed;
  var player=NewPlayer.instance;var bounds=RailWorld.Instance.boardingTrigger.bounds;
  originalStand=new Vector3(player.headCollider.transform.position.x,player.transform.position.y,player.headCollider.transform.position.z);
  float x=bounds.center.x+(main?-.45f:.45f);
  inPoint=new Vector3(x,player.transform.position.y,bounds.center.z);outPoint=new Vector3(x,player.transform.position.y,bounds.min.z-1.5f);
  ScanAudio(true);Warp(false);
  Log("INITIAL rest="+rest+" surface="+AudioSurface.TypeOf(plate.contactSurface)?.name+" collider="+center+" size="+size);
  if(main){start=PhotonNetwork.Time+3;mainActor=PhotonNetwork.LocalPlayer.ActorNumber;peerActor=PhotonNetwork.PlayerList.First(p=>!p.IsInactive&&p.ActorNumber!=mainActor).ActorNumber;
   File.WriteAllLines(Sync,new[]{room,start.ToString("R",CultureInfo.InvariantCulture),mainActor.ToString(),peerActor.ToString()});SetSync();}
 }
 void SetSync(){synced=true;syncReal=Time.realtimeSinceStartupAsDouble;syncAge=PhotonNetwork.Time-start;Log("SYNC start="+start.ToString("R",CultureInfo.InvariantCulture)+" age="+syncAge+" mainActor="+mainActor+" peerActor="+peerActor);}
 void Warp(bool value){inside=value;if(NewPlayer.instance!=null)NewPlayer.instance.Warp(value?inPoint:outPoint,Quaternion.identity);Log("CONTROLLED occupancy="+value+" target="+(value?inPoint:outPoint));}
 void ScanAudio(bool baseline){var am=AudioManager.instance;if(am==null)return;var voices=RailRemoteBridge.Get(am,"voices") as Array;if(voices==null)return;
  for(int i=0;i<voices.Length;i++){var v=voices.GetValue(i);var sound=RailRemoteBridge.Get(v,"Event") as SoundEvent;if(sound!=on&&sound!=off)continue;
   string key=am.GetInstanceID()+":"+i+":"+RailRemoteBridge.Get(v,"Generation");if(!seenAudio.Add(key)||baseline)continue;
   if(sound==on)ons++;else offs++;Log("AUDIO "+sound.name+" lease="+key+" On="+ons+" Off="+offs);}}
 void LateUpdate(){if(done)return;try{
  double real=Time.realtimeSinceStartupAsDouble;
  if(real-armed>65){Finish(false,"65-second bound reached; ready expedition not observed");return;}
  if(!synced){if(!File.Exists(Sync))return;var lines=File.ReadAllLines(Sync);if(lines.Length<4||lines[0]!=room||!double.TryParse(lines[1],NumberStyles.Float,CultureInfo.InvariantCulture,out var proposed))return;
   // Ignore old files. Peer-first injection waits for the new future start.
   if(PhotonNetwork.Time-proposed>.75||proposed-PhotonNetwork.Time>10)return;
   start=proposed;mainActor=int.Parse(lines[2]);peerActor=int.Parse(lines[3]);SetSync();}
  bool original=PhotonNetwork.InRoom&&PhotonNetwork.CurrentRoom.Name==room&&PhotonSession.Phase==RailPhase.Compound;
  double age=original?PhotonNetwork.Time-start:real-syncReal+syncAge;
  if(original&&!PhotonSession.Instance.HasRoomRequest){bool wanted=age>=12||(main?age>=0&&age<6:age>=3&&age<9);if(wanted!=inside)Warp(wanted);}
  double deadline=PhotonSession.Get(RailSessionRules.Deadline,0d);
  if(plate!=null&&plate.contactSurface.enabled&&plate.contactSurface.gameObject.activeInHierarchy){
   if(plate.Pressed!=pressed){pressed=plate.Pressed;edge=real;if(pressed)starts++;else stops++;Log("EDGE pressed="+pressed+" age="+age+" starts="+starts+" stops="+stops);}
   float drop=rest.y-plate.plate.localPosition.y;maxDrop=Mathf.Max(maxDrop,drop);
   moveError=Mathf.Max(moveError,Vector3.Distance(center,plate.contactSurface.bounds.center));sizeError=Mathf.Max(sizeError,Vector3.Distance(size,plate.contactSurface.bounds.size));rotationError=Mathf.Max(rotationError,Quaternion.Angle(rotation,plate.contactSurface.transform.rotation));
   Check(drop>=-.001f&&drop<=.031f,"plate motion stays within authored 3 cm travel");
   if(real-edge>.35){if(pressed&&Mathf.Abs(drop-.03f)<.001f)downSamples++;if(!pressed&&Mathf.Abs(drop)<.001f)upSamples++;}
  }
  ScanAudio(false);
  if(original&&!PhotonSession.Instance.HasRoomRequest){
   var roster=PhotonSession.Get(RailSessionRules.Roster,Array.Empty<int>());
   if(age>.8&&age<2.8){if(deadline>0){if(firstDeadline==0)firstDeadline=deadline;firstSeen=true;}Check(deadline>0,"first occupant starts countdown");Check(roster.Length==1&&roster.Contains(mainActor),"first roster contains main only");}
   if(age>3.8&&age<5.8){bothSeen=true;Check(roster.Length==2&&roster.Contains(mainActor)&&roster.Contains(peerActor),"second occupant joins roster");}
   if(age>6.8&&age<8.8){oneSeen=true;Check(roster.Length==1&&roster.Contains(peerActor),"main leaves while peer stays");}
   if(age>1&&age<8.8&&firstDeadline>0)Check(Math.Abs(deadline-firstDeadline)<.01,"deadline unchanged while at least one occupant remains");
   if(age>9.8&&age<11.8){cancelSeen=true;Check(deadline==0&&roster.Length==0,"last departure cancels countdown");Check(plate!=null&&!plate.Pressed&&Mathf.Abs(rest.y-plate.plate.localPosition.y)<.001f,"plate returns to rest after cancellation");Check(starts==1&&ons==1&&offs==1,"first cycle has exactly one On and one cancellation Off");}
   if(age>12.8&&age<31){if(secondDeadline==0&&deadline>0)secondDeadline=deadline;restartSeen=true;Check(deadline>0&&Math.Abs(deadline-secondDeadline)<.01,"restart countdown retained");Check(roster.Length==2&&roster.Contains(mainActor)&&roster.Contains(peerActor),"both occupants restart countdown");Check(starts==2&&ons==2&&offs==1,"no extra plate cue from occupancy changes");}
   if(age>0&&age<31&&real>=nextLog){nextLog=real+.5;Log("STATE age="+age.ToString("F3")+" localOn="+RailWorld.IsLocalOnPlatform()+" roster="+string.Join(",",roster)+" deadline="+deadline.ToString("F6")+" master="+PhotonNetwork.MasterClient.ActorNumber);}
   if(PhotonSession.Get(RailSessionRules.Destination,"").Length>0){departed=true;Check(secondDeadline>0&&PhotonNetwork.Time>=secondDeadline-.1,"full restarted countdown expires before transfer");}
  }
  if(!original&&age>=0){departed=true;Check(age>=31.5,"no premature room transfer");}
  var session=PhotonSession.Instance;
  if(session!=null&&session.WorldReady&&!session.HasRoomRequest&&PhotonSession.Phase==RailPhase.Running){Log("READY expedition room="+PhotonNetwork.CurrentRoom.Name+" run="+PhotonSession.RunId+" activeClients="+PhotonNetwork.PlayerList.Count(p=>!p.IsInactive));Finish(true,"ready expedition reached");}
 }catch(Exception ex){Log("FIXTURE ERROR "+ex);Finish(false,"fixture exception");}}
 void Finish(bool ready,string reason){if(done)return;done=true;
  if(!synced){Log("BLOCKED no current shared start received within bound");UnityEngine.Object.Destroy(gameObject);return;}
  Check(firstSeen&&bothSeen&&oneSeen&&cancelSeen&&restartSeen,"all five occupancy/deadline windows observed");
  Check(firstDeadline>0&&secondDeadline>firstDeadline+8&&Math.Abs(secondDeadline-(start+32))<1,"restarted countdown lasts normal 20 seconds");
  Check(starts==2&&ons==2&&offs>=1&&offs<=2,"two start On cues and one cancellation Off; optional departure Off only");
  Check(downSamples>0&&upSamples>0&&Mathf.Abs(maxDrop-.03f)<.001f,"plate reaches both authored endpoints");
  Check(moveError<.0001f&&sizeError<.0001f&&rotationError<.001f,"stationary contact collider");
  Log((failures.Count>0?"FAIL ":ready&&departed?"PASS ":"BLOCKED ")+"DONE "+reason+" failures="+failures.Count+" starts="+starts+" stops="+stops+" On="+ons+" Off="+offs+" maxDrop="+maxDrop+" colliderMove="+moveError+" colliderSize="+sizeError+" colliderRotation="+rotationError+" downSamples="+downSamples+" upSamples="+upSamples);
  if(PhotonNetwork.InRoom&&PhotonNetwork.CurrentRoom.Name==room&&PhotonSession.Phase==RailPhase.Compound&&NewPlayer.instance!=null)NewPlayer.instance.Warp(originalStand,Quaternion.identity);
  UnityEngine.Object.Destroy(gameObject);
 }
}
