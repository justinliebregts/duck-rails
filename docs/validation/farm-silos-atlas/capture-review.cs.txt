using UnityEngine; using UnityEditor; using UnityEngine.SceneManagement; using System; using System.Linq; using System.IO; using System.Text;
internal class CommandScript:IRunCommand {
Transform Resolve(Transform root,string key){foreach(var i in key.Split('/'))root=root.GetChild(int.Parse(i));return root;}
public void Execute(ExecutionResult result){
var root=SceneManager.GetActiveScene().GetRootGameObjects().Single(g=>g.GetComponent<RailBlock>());
if(!SceneManager.GetActiveScene().path.Contains("FarmSilosAtlasReview"))throw new Exception("Wrong scene");
var payload=JsonUtility.FromJson<RailFarmSilosAtlasReview.Payload>(File.ReadAllText("ArtSource/FarmSilosAtlas/unity-meshes.json"));
var originals=payload.sourceKeys.Select(k=>Resolve(root.transform,k).GetComponent<MeshRenderer>()).ToArray();
var chunks=root.transform.Find("Blender full FarmSilos - atlas review").GetComponentsInChildren<MeshRenderer>();
var overview=SceneManager.GetActiveScene().GetRootGameObjects().FirstOrDefault(g=>g.name.StartsWith("REVIEW ONLY - overview fill"));
bool overviewActive=overview&&overview.activeSelf;
if(overview)overview.SetActive(false);
var cameraObject=new GameObject("Temporary authoring capture");var cam=cameraObject.AddComponent<Camera>();
var fillObject=new GameObject("Temporary overview fill");var fill=fillObject.AddComponent<Light>();fill.type=LightType.Directional;fill.intensity=.8f;fill.shadows=LightShadows.None;fill.transform.rotation=Quaternion.Euler(40,-30,0);
cam.fieldOfView=70;cam.nearClipPlane=.05f;cam.farClipPlane=400;cam.clearFlags=CameraClearFlags.SolidColor;cam.backgroundColor=new Color(.12f,.13f,.15f);cam.allowHDR=false;
var rt=new RenderTexture(1280,720,24,RenderTextureFormat.ARGB32);cam.targetTexture=rt;
var previous=RenderTexture.active;bool fog=RenderSettings.fog;
var sb=new StringBuilder("Identical-camera authoring captures, fog disabled; overview has temporary directional fill. Other views use existing baked lighting. No Play Mode.\n");
string folder="docs/validation/farm-silos-atlas";
var names=new[]{"whole-block","rail-corridor","barn-interior","silo-grain","silo-ceiling","basement-cages","kitchen","attic","pond","corn","entrance-emission","distant-farm"};
var eyes=new[]{new Vector3(-110,120,-40),new Vector3(0,1.7f,45),new Vector3(26,1.5f,68),new Vector3(19.5f,8.2f,98),new Vector3(22,8,98),new Vector3(-27,-.7f,127),new Vector3(-27,2.3f,127),new Vector3(-26,8.5f,128),new Vector3(18,2,127),new Vector3(5,1,24),new Vector3(14,2,75),new Vector3(0,2,5)};
var targets=new[]{new Vector3(0,5,80),new Vector3(0,2,100),new Vector3(27,2,84),new Vector3(24,7.4f,99),new Vector3(22,12.8f,98.2f),new Vector3(-27,-.6f,139),new Vector3(-28,2.3f,140),new Vector3(-27,9,141),new Vector3(32,0,139),new Vector3(21,1.5f,35),new Vector3(17,3,75),new Vector3(0,4,100)};
try{
RenderSettings.fog=false;
for(int i=0;i<names.Length;i++){
cam.transform.position=eyes[i];cam.transform.rotation=Quaternion.LookRotation(targets[i]-eyes[i]);fill.enabled=i==0;
Color32[][] pixels=new Color32[2][];
for(int variant=0;variant<2;variant++){
foreach(var r in originals)r.enabled=variant==0;foreach(var r in chunks)r.enabled=variant==1;
cam.Render();RenderTexture.active=rt;var tex=new Texture2D(rt.width,rt.height,TextureFormat.RGB24,false);tex.ReadPixels(new Rect(0,0,rt.width,rt.height),0,0);tex.Apply();pixels[variant]=tex.GetPixels32();
File.WriteAllBytes(folder+"/"+names[i]+(variant==0?"-source.png":"-atlas.png"),tex.EncodeToPNG());UnityEngine.Object.DestroyImmediate(tex);
}
double error=0;int changed=0;for(int p=0;p<pixels[0].Length;p++){var a=pixels[0][p];var b=pixels[1][p];var delta=Math.Abs(a.r-b.r)+Math.Abs(a.g-b.g)+Math.Abs(a.b-b.b);error+=delta;if(delta>24)changed++;}
sb.AppendLine(names[i]+": mean RGB byte difference="+(error/(pixels[0].Length*3)).ToString("F3")+", pixels >8 average byte difference="+(100.0*changed/pixels[0].Length).ToString("F2")+"%");
}
}finally{foreach(var r in originals)r.enabled=false;foreach(var r in chunks)r.enabled=true;if(overview)overview.SetActive(overviewActive);RenderSettings.fog=fog;RenderTexture.active=previous;cam.targetTexture=null;UnityEngine.Object.DestroyImmediate(cameraObject);UnityEngine.Object.DestroyImmediate(fillObject);rt.Release();UnityEngine.Object.DestroyImmediate(rt);}
File.WriteAllText(folder+"/visual-comparison.txt",sb.ToString());result.Log(sb.ToString());
}
}
