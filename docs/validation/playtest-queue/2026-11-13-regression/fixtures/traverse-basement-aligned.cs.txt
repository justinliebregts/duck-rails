internal class CommandScript : IRunCommand {
 const string Path="X:/Work/duck-rails/docs/validation/playtest-queue/2026-11-13-regression/traversal-basement-aligned.txt";

 List<Route> routes=new List<Route>();int index,stage;double entered;NewPlayer p;SimulatedRigInput sim;Vector3 origin;Quaternion rotation;

No headset acceptance.\n");p=NewPlayer.instance;sim=SimulatedRigInput.Instance;origin=p.transform.position;rotation=p.transform.rotation;sim.SetSimulating(true);sim.UseInjectedInput=true;RailRemoteBridge.Set(sim,"yaw",0f);p.rb.isKinematic=false;

 foreach(var cell in new[]{new Vector3(18.5f,-3.2f,134),new Vector3(19.3f,-3.2f,139.65f),new Vector3(19.3f,-3.2f,143.2f),new Vector3(26.5f,-3.2f,128.7f),new Vector3(28,-3.2f,142.85f)}){
 Add("farm-silos-v1","basement cell enter "+cell,RailFarmSilos.HousePoint(cell+Vector3.back*.7f),RailFarmSilos.HousePoint(cell+Vector3.forward*.7f));

 void Tick(){if(!Application.isPlaying){EditorApplication.update-=Tick;return;}try{RailRemoteBridge.Set(UnityEngine.Object.FindFirstObjectByType<PhotonMasterHandoff>(),"focused",true);var q=routes[index];double age=EditorApplication.timeSinceStartup-entered;if(stage==0){if(age<.7)return;stage=1;entered=EditorApplication.timeSinceStartup;}
 Vector3 h=p.headCollider.transform.position,delta=q.b-h;delta.y=0;if(delta.magnitude<.3f&&stage==1){stage=2;entered=EditorApplication.timeSinceStartup;sim.InjectedInput=new SimulatedRigInput.Frame{leftReach=false,rightReach=false};return;}if(stage==2&&age<8)return;if(stage==2||age>30){Log((delta.magnitude<.65f&&sim.HasBodySupport&&h.y-q.b.y>.15f&&h.y-q.b.y<1.2f?"PASS ":"INCOMPLETE ")+q.label+" head="+h+" remainingXZ="+delta.magnitude+" target="+q.b+" down="+RailExpedition.LocalDown+" supported="+sim.HasBodySupport);index++;Start();return;}var local=p.transform.InverseTransformDirection(delta.normalized*.2f);sim.InjectedInput=new SimulatedRigInput.Frame{move=new Vector2(local.x,local.z),leftReach=false,rightReach=false,jump=sim.HasBodySupport&&age>1&&!q.label.Contains("basement")&&!q.label.Contains("ditch")&&!q.label.Contains("playground")};
 }catch(Exception e){Log("FIXTURE ERROR "+e);sim.InjectedInput=default;sim.UseInjectedInput=false;EditorApplication.update-=Tick;}}
}
