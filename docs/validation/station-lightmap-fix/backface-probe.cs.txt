using UnityEngine;using UnityEditor;using System.Collections.Generic;using System.Linq;internal class CommandScript:IRunCommand{struct Tri{public Vector3 a,e1,e2;public string name;}public void Execute(ExecutionResult r){var all=new List<Tri>();foreach(var f in Object.FindObjectsByType<MeshFilter>(FindObjectsSortMode.None)){var rr=f.GetComponent<MeshRenderer>();if(!rr||!rr.enabled||!GameObjectUtility.AreStaticEditorFlagsSet(f.gameObject,StaticEditorFlags.ContributeGI)||!f.sharedMesh)continue;var m=f.sharedMesh;var v=m.vertices.Select(p=>f.transform.TransformPoint(p)).ToArray();var t=m.triangles;for(int i=0;i<t.Length;i+=3)all.Add(new Tri{a=v[t[i]],e1=v[t[i+1]]-v[t[i]],e2=v[t[i+2]]-v[t[i]],name=f.name});}foreach(float xx in new[]{-8f,8f})foreach(float zz in new[]{72f,80f,88f}){var origin=new Vector3(xx,1.01f,zz);int front=0,back=0,miss=0;var backs=new Dictionary<string,int>();for(int i=0;i<128;i++){float y=(i+.5f)/128;float a=i*2.39996323f;var d=new Vector3(Mathf.Sqrt(1-y*y)*Mathf.Cos(a),y,Mathf.Sqrt(1-y*y)*Mathf.Sin(a));float nearest=10000;bool isBack=false;string name="";foreach(var q in all){var h=Vector3.Cross(d,q.e2);float det=Vector3.Dot(q.e1,h);if(Mathf.Abs(det)<1e-7)continue;float inv=1/det;var s=origin-q.a;float u=inv*Vector3.Dot(s,h);if(u<0||u>1)continue;var z=Vector3.Cross(s,q.e1);float w=inv*Vector3.Dot(d,z);if(w<0||u+w>1)continue;float dist=inv*Vector3.Dot(q.e2,z);if(dist>.0001f&&dist<nearest){nearest=dist;isBack=det<0;name=q.name;}}if(nearest==10000){miss++;continue;}if(isBack){back++;if(!backs.ContainsKey(name))backs[name]=0;backs[name]++;}else front++;}r.Log(origin+" front="+front+" back="+back+" miss="+miss+" backObjects="+string.Join(",",backs.Select(k=>k.Key+":"+k.Value)));}}}
