// Execute ONLY on ready idle NONMASTER of two-client private run. HARD15s, normally13s.
// Ordinary simulated grip edge is fed BEFORE SimulatedRigInput.Update; raw palm positioning
// follows afterward. Never bypass shared lease guard or change authored0.25s return duration.
// Regrip missing the actual shared-clear window is BLOCKED, not forced/retried.
// Independent second hold: explicitly inject legacy Tab panel's open UI state (RailWorld.menu),
// observe production blocked-input cleanup, then restore. This is NOT Tab/P hardware delivery,
// NOT PauseMenu-open STOP, NOT down/disconnect coverage. No gameplay counter/state seeding.
using System;using System.IO;using System.Linq;using System.Collections.Generic;
using UnityEngine;using Photon.Pun;using Game.Animation;
internal class CommandScript:IRunCommand {
 public void Execute(ExecutionResult r){
  if(!Application.isPlaying||!PhotonNetwork.InRoom||PhotonNetwork.IsMasterClient||PhotonNetwork.CurrentRoom.PlayerCount!=2||!PhotonSession.IsPrivate||PhotonSession.Phase!=RailPhase.Running||!PhotonSession.Instance.WorldReady)throw new Exception("Need idle ready nonmaster in two-client private run");
  if(GameObject.Find("Narrow throttle regrip panel")!=null)throw new Exception("Probe already active");
  var go=new GameObject("Narrow throttle regrip panel");var owner=go.AddComponent<QueueNarrowThrottle>();go.AddComponent<QueueNarrowThrottlePose>().owner=owner;go.AddComponent<QueueNarrowThrottleObserve>().owner=owner;r.RegisterObjectCreation(go);r.Log("Started15s narrow regrip/panel probe; no .25s tuning change, no source edits");
 }
}
[DefaultExecutionOrder(-16300)] internal class QueueNarrowThrottle:MonoBehaviour {
 RailControl drive;SimulatedRigInput sim;NewPlayer motor;LocalBodyDriver body;RailWorld world;PlayerHand hand;
 string path,run,room,firstToken;int actor,checks,failures;double began,releaseObservedAt,menuAcquireAt;
 bool done,saved,requested,regripped,blocked,regripReported,firstHeldChecked,menuInjected,menuPassed,menuClosed,releaseSeen;
 bool oldInject,oldDisable,oldKinematic,oldMenu,oldCursor;CursorLockMode oldLock;SimulatedRigInput.Frame oldFrame;
 Vector3 origin,cabin,offPalm,velocity,angular;Quaternion heading;float wantedTravel,regripRaw;bool wantedGrip;
 readonly Dictionary<string,object> poses=new Dictionary<string,object>();
 T Get<T>(object x,string f)=>(T)RailRemoteBridge.Get(x,f);
 float Progress=>Get<float>(drive,"renderedProgress");float Target=>Convert.ToSingle(PhotonSession.Get(RailSessionRules.Motion,new object[]{0d,0f,0f,0f})[3]);
 Transform Palm=>body.RightPalmSource;
 void Log(string s)=>File.AppendAllText(path,DateTime.UtcNow.ToString("O")+" "+s+"\n");
 void Check(bool ok,string s){checks++;Log((ok?"PASS ":"FAIL ")+s);if(!ok)throw new Exception(s);}
 void Start(){try{
  actor=PhotonNetwork.LocalPlayer.ActorNumber;room=PhotonNetwork.CurrentRoom.Name;run=PhotonSession.RunId;
  string root=Application.dataPath.Replace('\\','/');int vp=root.IndexOf("/Library/VP/",StringComparison.OrdinalIgnoreCase);root=vp<0?Path.GetDirectoryName(root):root.Substring(0,vp);path=root+"/docs/validation/playtest-queue/2026-11-16/throttle-narrow-actor"+actor+".txt";
  File.WriteAllText(path,"Narrow default-return regrip with pre-input scheduling; actual grip/lease path. Legacy Tab panel open-state injection for blocked-input cleanup; no hardware/PauseMenu claim.\n");
  drive=UnityEngine.Object.FindObjectsByType<RailControl>(FindObjectsSortMode.None).Single(x=>x.action==RailControl.Action.Drive);sim=SimulatedRigInput.Instance;motor=NewPlayer.instance;body=motor.GetComponent<LocalBodyDriver>();hand=GR.Player.instance.rightPlayerHand;world=RailWorld.Instance;
  Check(sim.IsSimulating&&!sim.UseInjectedInput&&!sim.Buttons.grip&&sim.Buttons.move==Vector2.zero&&sim.Buttons.look==Vector2.zero&&!sim.ThrottleHeld&&!hand.IsHolding&&!GR.Player.instance.leftPlayerHand.IsHolding&&!RailWorld.InputBlocked&&!RailExpedition.LocalDown,"neutral idle actual rig");
  Check(Mathf.Abs(drive.returnDuration-.25f)<.0001f&&PhotonSession.DriveGripActor==0&&Mathf.Abs(Target)<.001f&&RailTrain.Instance.Speed<.01f,"authored0.25s control initially unowned/stopped");
  oldInject=sim.UseInjectedInput;oldFrame=sim.InjectedInput;oldDisable=motor.disableMovement;oldKinematic=motor.rb.isKinematic;velocity=motor.rb.linearVelocity;angular=motor.rb.angularVelocity;oldMenu=Get<bool>(world,"menu");oldCursor=Cursor.visible;oldLock=Cursor.lockState;
  origin=RailTrain.Instance.transform.InverseTransformPoint(new Vector3(motor.headCollider.transform.position.x,motor.transform.position.y,motor.headCollider.transform.position.z));heading=Quaternion.Inverse(RailTrain.Instance.transform.rotation)*motor.transform.rotation;
  foreach(string n in new[]{"headPosition","leftPosition","rightPosition","headRotation","leftRotation","rightRotation","yaw","pitch","reach","handRoll","awaitNeutral","awaitThrottleNeutral","throttleKeyWasHeld","Buttons","ThrottleHeld"})poses[n]=RailRemoteBridge.Get(sim,n);
  saved=true;var stand=drive.transform.position-RailTrain.Instance.transform.right*.12f-RailTrain.Instance.transform.forward*.65f;stand.y=motor.transform.position.y;motor.Warp(stand,RailTrain.Instance.transform.rotation);cabin=RailTrain.Instance.transform.InverseTransformPoint(motor.transform.position);motor.disableMovement=true;motor.rb.isKinematic=true;sim.UseInjectedInput=true;
  offPalm=drive.transform.InverseTransformPoint(drive.handle.position+Vector3.back*.025f);began=Time.realtimeSinceStartupAsDouble;
 }catch(Exception e){Fail(e);}}
 void Update(){if(done||!saved)return;try{
  double age=Time.realtimeSinceStartupAsDouble-began;if(age>15)throw new Exception("15-second bound");if(!PhotonNetwork.InRoom||PhotonNetwork.CurrentRoom.Name!=room||PhotonSession.RunId!=run||PhotonSession.Phase!=RailPhase.Running||RailExpedition.LocalDown)throw new Exception("Room/run/local availability changed");
  motor.transform.position=RailTrain.Instance.transform.TransformPoint(cabin);wantedGrip=false;wantedTravel=Progress;
  if(age<.5)wantedTravel=0;
  else if(age<3.5){wantedGrip=true;wantedTravel=Mathf.Clamp01((float)(age-1.5));}
  else if(age<5.5){
   if(!requested&&!drive.HeldBy(hand)&&PhotonSession.DriveGripActor==0&&Get<bool>(drive,"returning")&&Progress>.001f){requested=true;regripRaw=Progress;Log("INPUT first eligible ordinary regrip BEFORE SimulatedRigInput.Update progress="+Progress+" age="+age+" sharedActor=0");}
   wantedGrip=requested;wantedTravel=regripped?regripRaw:Progress;
  }
  else if(age>=6.5&&age<9.5){
   if(menuAcquireAt==0&&PhotonSession.DriveGripActor==0&&!drive.HeldBy(hand)){menuAcquireAt=age;Log("INPUT independent second ordinary grip for panel cleanup");}
   wantedGrip=menuAcquireAt>0;wantedTravel=menuAcquireAt>0?Mathf.Clamp01((float)(age-menuAcquireAt-.5)):Progress;
  }
  else if(age>=9.5&&age<11.5){
   wantedGrip=true;wantedTravel=1;
   if(!menuInjected){Check(drive.HeldBy(hand)&&PhotonSession.DriveGripActor==actor&&Target>.1f,"independent actual grip/GO ready for blocked-input check");menuInjected=true;RailRemoteBridge.Set(world,"menu",true);Log("INPUT controlled legacy Tab-panel open-state injection; no physical Tab key, no PauseMenu forced STOP");}
  }
  if(age>=11.5&&!menuClosed){menuClosed=true;RailRemoteBridge.Set(world,"menu",oldMenu);Cursor.visible=oldCursor;Cursor.lockState=oldLock;Log("RESTORE legacy panel state; grip input released before unblocking");}
  sim.InjectedInput=new SimulatedRigInput.Frame{grip=wantedGrip,rightReach=true};
  if(age>13){Check(menuPassed&&menuClosed&&PhotonSession.DriveGripActor==0&&Mathf.Abs(Target)<.001f,"final shared state stopped and panel restored");Finish();}
 }catch(Exception e){Fail(e);}}
 public void Pose(){if(done||!saved)return;try{Vector3 target=drive.transform.TransformPoint(Quaternion.Euler(wantedTravel*70,0,0)*offPalm);Vector3 offset=Palm.position-body.RightHandAnchor.position;RailRemoteBridge.Set(sim,"rightPosition",body.RightHandAnchor.parent.InverseTransformPoint(target-offset));RailRemoteBridge.Call(sim,"WriteSimulatedPose");}catch(Exception e){Fail(e);}}
 public void Observe(){if(done||!saved)return;try{double age=Time.realtimeSinceStartupAsDouble-began;
  if(age>3&&age<3.5&&!firstHeldChecked){firstHeldChecked=true;Check(drive.HeldBy(hand)&&PhotonSession.DriveGripActor==actor&&Target>.1f,"first ordinary grip has granted live GO");firstToken=Get<string>(drive,"gripToken");}
  if(age>=3.5&&age<4.6){
   if(!drive.HeldBy(hand)&&!releaseSeen){releaseSeen=true;releaseObservedAt=Time.realtimeSinceStartupAsDouble;}
   Log("WINDOW dt="+(releaseSeen?Time.realtimeSinceStartupAsDouble-releaseObservedAt:-1)+" progress="+Progress+" sharedActor="+PhotonSession.DriveGripActor+" held="+drive.HeldBy(hand)+" requested="+requested+" inputGrip="+sim.Buttons.grip);
   if(requested&&!regripped&&drive.HeldBy(hand)&&Get<string>(drive,"gripToken")!=firstToken){float captured=Get<float>(drive,"gripStartProgress");
    if(captured<=.001f){blocked=true;Log("BLOCKED ordinary grip landed after default return completed");}else{Check(captured<1&&Mathf.Abs(captured-Progress)<.0001f,"default-window grip captures current rendered angle captured="+captured+" rendered="+Progress);regripped=true;regripRaw=wantedTravel;}
   }
  }
  if(age>5.2&&!regripReported){regripReported=true;if(regripped)Check(drive.HeldBy(hand)&&PhotonSession.DriveGripActor==actor&&(string)PhotonSession.DriveGripState[2]==Get<string>(drive,"gripToken"),"default-window regrip receives actual new shared lease");else{blocked=true;Log("BLOCKED no successful default-window regrip; shared stale-own lease cannot be bypassed");}}
  if(age>10.9&&!menuPassed){Check(menuInjected&&RailWorld.InputBlocked&&!drive.HeldBy(hand)&&!RailControl.TryGetDrivePalm(Palm,out _,out _)&&PhotonSession.DriveGripActor==0&&Mathf.Abs(Target)<.001f,"real blocked-input panel state releases actual attached hand/lease and stops target");menuPassed=true;}
 }catch(Exception e){Fail(e);}}
 void Fail(Exception e){if(done)return;failures++;if(path!=null)Log("FAIL "+e);Finish();}
 void Finish(){if(done)return;done=true;if(saved){drive.ReleaseDrive();hand.Release();RailRemoteBridge.Set(world,"menu",oldMenu);Cursor.visible=oldCursor;Cursor.lockState=oldLock;foreach(var p in poses)RailRemoteBridge.Set(sim,p.Key,p.Value);RailRemoteBridge.Call(sim,"WriteSimulatedPose");sim.UseInjectedInput=oldInject;sim.InjectedInput=oldFrame;motor.disableMovement=oldDisable;motor.rb.isKinematic=oldKinematic;if(RailTrain.Instance!=null)motor.Warp(RailTrain.Instance.transform.TransformPoint(origin),RailTrain.Instance.transform.rotation*heading);if(!oldKinematic){motor.rb.linearVelocity=velocity;motor.rb.angularVelocity=angular;}}
  if(path!=null)Log("END checks="+checks+" failures="+failures+" defaultRegrip="+regripped+" regripBlocked="+blocked+" panelCleanup="+menuPassed+"; restored rig/input/panel/cursor; default duration untouched");Destroy(gameObject);}
 void OnDestroy(){if(!done)Finish();}
}
