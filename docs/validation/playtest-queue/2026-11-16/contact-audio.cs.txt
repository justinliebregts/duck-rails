// Run in existing desktop-simulated Play, with local hands empty and no other movement fixture.
// REAL body falls through NewPlayer.OnCollisionEnter; INJECTED hand callback through NewPlayer.PlayHandHit.
// Temporarily moves the local rig; restores stand point (train-relative if originally aboard), view/input.
// Runs fountain cases when present, Farm cases when present; neither case creates surfaces or audio events.
using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using UnityEngine;using Game.Audio;
internal class CommandScript:IRunCommand {
 public void Execute(ExecutionResult r){if(!Application.isPlaying||!NewPlayer.instance||!SimulatedRigInput.Instance||!SimulatedRigInput.Instance.IsSimulating)throw new Exception("Need existing desktop simulated local rig");if(RailWorld.InputBlocked||RailExpedition.LocalDown)throw new Exception("Need unblocked alive local rig");if(UnityEngine.Object.FindFirstObjectByType<QueueContactAudio>())throw new Exception("Contact probe already running");var g=new GameObject("Queue contact audio");g.AddComponent<QueueContactAudio>();r.RegisterObjectCreation(g);r.Log("Started bounded actual body falls + explicitly injected hand callback; restores rig; appends contact-audio-actorN.txt");}
}
internal class QueueContactAudioTap:MonoBehaviour {
 public QueueContactAudio owner;
 void OnCollisionEnter(Collision c){if(owner)owner.Contact(c);}
}
[DefaultExecutionOrder(32000)]internal class QueueContactAudio:MonoBehaviour {
 class Case {public string name,kind;public RaycastHit hit;}
 class Heard {public SoundEvent e;public Vector3 p;public float t;public string key;}
 class Impact {public Collider collider;public Vector3 p;public float t,speed;public SoundEvent expected;}
 readonly List<Case> cases=new List<Case>();readonly List<Heard> heard=new List<Heard>();readonly List<Impact> impacts=new List<Impact>();readonly HashSet<string> seen=new HashSet<string>();
 NewPlayer p;SimulatedRigInput sim;QueueContactAudioTap tap;string output,stage="idle";int pass,fail,blocked;bool saved,restored,done;Vector3 stand,trainLocal;Quaternion facing;Transform train;float pitch,reach;bool injected;SimulatedRigInput.Frame input;
 void Log(string s)=>File.AppendAllText(output,DateTime.UtcNow.ToString("O")+" "+s+"\n");
 void Check(bool ok,string s){if(ok)pass++;else fail++;Log((ok?"PASS ":"FAIL ")+s);}
 void Block(string s){blocked++;Log("BLOCKED "+s);}
 void Sample(string name,Vector3 above,string kind,Transform requiredRoot=null){if(!Physics.Raycast(above,Vector3.down,out var h,5,1<<6,QueryTriggerInteraction.Ignore)){Block(name+" no Environment hit");return;}if(requiredRoot&&!h.collider.transform.IsChildOf(requiredRoot)){Block(name+" hits other block "+h.collider.name);return;}var water=AudioSurface.TypeAt(h.collider,h.point);var foliage=FoliageVolume.Find(h.point);bool valid=kind=="Water"?water&&water.name=="Water":kind=="Foliage"?foliage!=null:foliage==null&&(!water||water.name!="Water");if(!valid){Block(name+" candidate does not resolve requested "+kind+" surface="+water?.name+" foliage="+foliage?.name);return;}cases.Add(new Case{name=name,kind=kind,hit=h});Log("TARGET "+name+" point="+h.point+" collider="+h.collider.name+" surface="+water?.name+" foliage="+foliage?.name+" body="+p.bodyLandingSounds?.Resolve(h.collider,h.point)?.name+" hand="+p.handHitSounds?.Resolve(h.collider,h.point)?.name);}
 void Build(){var water=UnityEngine.Object.FindObjectsByType<BoxCollider>(FindObjectsSortMode.None).FirstOrDefault(c=>c.name=="Water audio collider");if(water){foreach(float x in new[]{-.85f,.85f})Sample("Fountain basin "+x,water.transform.TransformPoint(water.center+new Vector3(x,0,.85f))+Vector3.up*2,"Water");Sample("Fountain outside",water.transform.TransformPoint(water.center+new Vector3(2.8f,0,0))+Vector3.up*2,"Dry");}else Log("NOT_PRESENT Compound fountain (run again in Compound)");var farm=UnityEngine.Object.FindObjectsByType<RailBlock>(FindObjectsSortMode.None).FirstOrDefault(b=>b.id=="farm-silos-v1");if(farm){int n=0;foreach(var v in farm.GetComponentsInChildren<FoliageVolume>().Where(v=>v.stalks.Length>0)){var s=v.stalks[v.stalks.Length/2];int before=cases.Count;Sample("Farm stalk patch "+n,v.transform.TransformPoint(new Vector3(s.x,s.y,s.z))+Vector3.up*2,"Foliage",farm.transform);if(cases.Count>before&&++n==2)break;}if(n<2)Block("Fewer than two valid Farm stalk contact points");Sample("Farm dry railway-side ground",farm.transform.TransformPoint(new Vector3(6,2,50)),"Dry",farm.transform);Sample("Farm ditch water priority",farm.transform.TransformPoint(new Vector3(12,2,30)),"Water",farm.transform);}else Log("NOT_PRESENT Farm (run again with Farm loaded)");}
 void Capture(){var am=AudioManager.instance;if(!am)return;var voices=RailRemoteBridge.Get(am,"voices") as IEnumerable;if(voices==null)return;foreach(var v in voices){var ev=RailRemoteBridge.Get(v,"Event") as SoundEvent;var source=RailRemoteBridge.Get(v,"Source") as AudioSource;if(!ev||!source)continue;string key=am.GetInstanceID()+":"+source.GetInstanceID()+":"+RailRemoteBridge.Get(v,"Generation");if(!seen.Add(key))continue;var h=new Heard{e=ev,p=source.transform.position,t=Time.unscaledTime,key=key};heard.Add(h);if(stage!="idle")Log("LEASE stage="+stage+" event="+ev.name+" point="+h.p+" key="+key+" playing="+source.isPlaying+" volume="+source.volume);}}
 void LateUpdate(){if(output!=null)Capture();}
 public void Contact(Collision c){if(stage!="body"||!p||c.gameObject.layer!=6)return;for(int i=0;i<c.contactCount;i++){var a=c.GetContact(i);if(a.thisCollider!=p.bodyCollider||a.normal.y<=.5f)continue;float speed=c.relativeVelocity.magnitude;var expected=p.bodyLandingSounds?p.bodyLandingSounds.Resolve(a.otherCollider,a.point):null;Log("ACTUAL BODY COLLISION collider="+a.otherCollider.name+" point="+a.point+" normal="+a.normal+" speed="+speed+" expected="+expected?.name);if(speed>=2)impacts.Add(new Impact{collider=a.otherCollider,p=a.point,t=Time.unscaledTime,speed=speed,expected=expected});break;}}
 IEnumerator Wait(float seconds){float end=Time.realtimeSinceStartup+seconds;while(Time.realtimeSinceStartup<end)yield return null;}
 IEnumerator Start(){output="X:/Work/duck-rails/docs/validation/playtest-queue/2026-11-16/contact-audio-actor"+Photon.Pun.PhotonNetwork.LocalPlayer.ActorNumber+".txt";Log("BEGIN Unity="+Application.unityVersion+" real local NewPlayer.bodyLandingSounds collision path; hand callback INJECTED, never direct AudioManager.Play. Dispatch/source evidence only, no listening, cadence, remote footsteps or natural VR acceptance.");p=NewPlayer.instance;sim=SimulatedRigInput.Instance;if(!AudioManager.instance||!p.bodyLandingSounds||!p.handHitSounds){Block("Missing live manager/bodyLandingSounds/handHitSounds");done=true;Destroy(gameObject);yield break;}
  stand=p.headCollider.transform.position;stand.y=p.transform.position.y;facing=p.headCollider.transform.rotation;train=RailTrain.Instance&&RailTrain.Instance.Contains(p.headCollider.transform.position)?RailTrain.Instance.transform:null;if(train)trainLocal=train.InverseTransformPoint(stand);pitch=(float)RailRemoteBridge.Get(sim,"pitch");reach=(float)RailRemoteBridge.Get(sim,"reach");injected=sim.UseInjectedInput;input=sim.InjectedInput;saved=true;tap=p.gameObject.AddComponent<QueueContactAudioTap>();tap.owner=this;
  try{Build();foreach(var q in cases){var hs=p.handHitSounds.Resolve(q.hit.collider,q.hit.point);var bs=p.bodyLandingSounds.Resolve(q.hit.collider,q.hit.point);if(q.kind!="Dry")Check(hs&&bs&&hs.name=="HandHit_"+q.kind&&bs.name=="HandHit_"+q.kind,q.name+" current hand/body mappings are HandHit_"+q.kind);else Check(hs&&bs&&hs.name!="HandHit_Water"&&hs.name!="HandHit_Foliage"&&bs.name!="HandHit_Water"&&bs.name!="HandHit_Foliage",q.name+" dry control restores non-water/non-foliage mappings");}if(cases.Count==0){Block("No valid current scene target");yield break;}sim.UseInjectedInput=true;sim.InjectedInput=new SimulatedRigInput.Frame{leftReach=true,rightReach=true};RailRemoteBridge.Set(sim,"pitch",-70f);RailRemoteBridge.Set(sim,"reach",1f);Capture();
   foreach(var q in cases){if(!p||!q.hit.collider){Block(q.name+" scene/player changed");break;}stage="idle";p.Warp(q.hit.point+Vector3.up*.8f);yield return null;heard.Clear();impacts.Clear();Capture();heard.Clear();stage="body";p.rb.linearVelocity=Vector3.down*3f;Log("DROP "+q.name+" real body; scripted start/velocity, normal movement/collision callbacks enabled");yield return Wait(2);Capture();stage="idle";
    var valid=impacts.Where(a=>a.collider==q.hit.collider&&Vector3.Distance(a.p,q.hit.point)<.65f).ToArray();if(valid.Length==0)Block(q.name+" no qualifying >=2m/s actual body collision on target; fixture/drop incomplete, not sound failure");else foreach(var a in valid){if(!a.expected){Block(q.name+" body sound mapping null");continue;}int matched=heard.Count(h=>h.e==a.expected&&Mathf.Abs(h.t-a.t)<.16f&&Vector3.Distance(h.p,a.p)<.3f);Check(matched>0,q.name+" actual body landing dispatch="+a.expected.name+" matchingLeases="+matched+" impact="+a.speed);}
    // Exercise actual hand selection/dispatch callback; this does NOT exercise hand solver/cadence/haptic gates.
    var expected=p.handHitSounds.Resolve(q.hit.collider,q.hit.point);Capture();heard.Clear();stage="injected-hand";Log("INJECTED PlayHandHit collider="+q.hit.collider.name+" point="+q.hit.point+" volume=.6 target="+q.name);RailRemoteBridge.Call(p,"PlayHandHit",q.hit.collider,q.hit.point,.6f);Capture();yield return Wait(.2f);Capture();int count=heard.Count(h=>h.e==expected&&Vector3.Distance(h.p,q.hit.point)<.25f);Check(expected&&count==1,q.name+" INJECTED hand callback dispatch expected="+expected?.name+" matchingLeases="+count);stage="idle";yield return Wait(.4f);
   }done=true;Log("DONE pass="+pass+" fail="+fail+" blocked="+blocked+"; actual body collision and INJECTED hand cases are separate acceptance scopes");
  }finally{Restore();Destroy(gameObject);}
 }
 void Restore(){if(restored)return;restored=true;stage="idle";if(tap){tap.owner=null;Destroy(tap);}if(saved&&sim){sim.InjectedInput=input;sim.UseInjectedInput=injected;RailRemoteBridge.Set(sim,"pitch",pitch);RailRemoteBridge.Set(sim,"reach",reach);}if(saved&&p){p.Warp(train?train.TransformPoint(trainLocal):stand,facing);p.rb.linearVelocity=Vector3.zero;}if(output!=null)Log("RESTORED local stand/input/view; sounds end normally");}
 void OnDestroy(){Restore();if(!done&&output!=null)Log("ABORTED or fixture exception: inspect Console; no completion acceptance");}
}
