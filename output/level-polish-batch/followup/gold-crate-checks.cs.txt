using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.Rendering;using System.Linq;using System.IO;using System.Collections.Generic;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult r){RailCanyon.Guard();var lines=new List<string>();int count=0;void Check(bool value,string name){count++;if(!value)throw new System.Exception(name);lines.Add("PASS "+name);}
var asset=AssetDatabase.LoadAssetAtPath<GameObject>(RailFinaleFort.PrefabPath);var t=asset.GetComponentsInChildren<Transform>(true).Single(t=>t.name=="Finale gold deposit");var f=t.GetComponentsInChildren<MeshFilter>().Single(f=>f.name=="CrateGoldBarMesh");var mr=f.GetComponent<MeshRenderer>();var collision=f.GetComponent<MeshCollider>().sharedMesh;
Check(f.sharedMesh!=collision,"dedicated render mesh; original collision mesh retained");Check(AssetDatabase.GetAssetPath(collision)=="Assets/Models/CrateGoldBar/CrateGoldBar.fbx","source FBX retained for collision");Check(collision.uv2.Length==0,"shared FBX unchanged");
Check(f.sharedMesh.uv2.Length==f.sharedMesh.vertexCount,"complete lightmap UVs");Check(f.sharedMesh.vertexCount==collision.vertexCount&&f.sharedMesh.triangles.Length==collision.triangles.Length,"geometry counts unchanged");Check(f.sharedMesh.bounds==collision.bounds,"geometry bounds unchanged");
Check(GameObjectUtility.AreStaticEditorFlagsSet(f.gameObject,StaticEditorFlags.ContributeGI),"stationary mesh contributes GI");Check(mr.receiveGI==ReceiveGI.Lightmaps&&mr.lightProbeUsage==LightProbeUsage.Off,"stationary mesh uses lightmaps");Check(asset.GetComponent<RailBakedBlockLighting>().bindings.Any(b=>b.renderer==mr&&b.color),"crate exported lightmap binding present");
var receiver=t.GetComponent<RailGoldReceiver>();Check(receiver.receiverId==3&&receiver.intake&&receiver.coins&&receiver.label,"deposit references retained");Check(!receiver.goldTextSpawn&&!receiver.goldTextFont&&!receiver.goldTextMaterial,"pre-existing unassigned reward-text fields unchanged (confirmed against baseline)");Check(!GameObjectUtility.AreStaticEditorFlagsSet(receiver.label.gameObject,StaticEditorFlags.ContributeGI),"dynamic text not marked GI static");Check(receiver.intake.isTrigger,"intake remains trigger");
RailCatalogLightingChecks.InstallFinale();var target=Object.FindFirstObjectByType<RailRoute>().finaleRoot;var cr=target.GetComponentsInChildren<MeshRenderer>().Single(x=>x.name=="CrateGoldBarMesh");Check(target.GetComponent<RailBakedBlockLighting>().bindings.Any(b=>b.renderer==cr&&b.color),"crate binding installed in Expedition");EditorSceneManager.OpenScene(RailFinaleFort.BakeScene);var live=Object.FindFirstObjectByType<RailBlock>().GetComponentsInChildren<MeshRenderer>().Single(x=>x.name=="CrateGoldBarMesh");Check(live.lightmapIndex>=0&&live.lightmapIndex<LightmapSettings.lightmaps.Length,"reopened authoring crate has atlas assignment");Check(!EditorUtility.scriptCompilationFailed,"compilation clean");Check(!EditorApplication.isPlaying,"no Play Mode");
lines.Insert(0,count+" editor checks passed. Crate material/transform/behaviour unchanged; new lightmap UVs and static GI only.");File.WriteAllLines("output/level-polish-batch/followup/gold-crate-checks.txt",lines);r.Log(string.Join("\n",lines));}}
