internal class CommandScript:IRunCommand{
 public void Execute(ExecutionResult r){
  if(!Application.isPlaying||!PhotonSession.IsPrivate||PhotonSession.Phase!=RailPhase.Running||PhotonSession.Instance==null||!PhotonSession.Instance.WorldReady||RailTrain.Instance==null)
   throw new Exception("Need ready private Running clients, master drives and other clients observe");
  if(RailTrain.Instance.Speed>.01f||Mathf.Abs(Convert.ToSingle(PhotonSession.Get(RailSessionRules.Motion,new object[]{0d,0f,0f,0f})[3]))>.01f)
   throw new Exception("Need stopped train with retained target speed zero; no distance reset is performed");
  var motion=PhotonSession.Get(RailSessionRules.Motion,new object[]{0d,0f,0f,0f});
  if(Mathf.Abs(Convert.ToSingle(motion[2]))>.001f||Mathf.Abs(Convert.ToSingle(motion[1])-RailTrain.Instance.Distance)>.01f)
   throw new Exception("Need zero-speed motion anchor at current distance (fresh run); fuel-only writes must not reinterpret old deceleration");
  if(PhotonNetwork.IsMasterClient&&(!RailTestTools.CanMutate||RailDiagnosticPanel.Instance==null||NewPlayer.instance==null||RailExpedition.LocalDown||SimulatedRigInput.Instance==null||!SimulatedRigInput.Instance.IsSimulating||SimulatedRigInput.Instance.UseInjectedInput))
   throw new Exception("Master needs available test tools, live local Keyboard rig, no competing injected input");
  var g=new GameObject("Bounded horn fuel UI evidence");g.AddComponent<HornFuelQueue>();r.RegisterObjectCreation(g);r.Log("Horn/fuel observer armed; inject peers first, then master. 65-second bound.");
 }
}

   revision=PhotonSession.Get(RailSessionRules.Revision,0);retry=began+.5;

 void SampleFuel(float value){
  int expected=Mathf.CeilToInt(Mathf.Clamp01(value/bar.capacity)*10),indices=bar.segments.sharedMesh.triangles.Length;var gauge=RailExpedition.Instance.gauge;
  string text=gauge==null?"":gauge.GetParsedText();int label=text.IndexOf("FUEL",StringComparison.Ordinal);string token="";if(label>=0){int at=label+4;while(at<text.Length&&char.IsWhiteSpace(text[at]))at++;while(at<text.Length&&(char.IsDigit(text[at])||text[at]=='.'))token+=text[at++];}float shown;bool parsed=float.TryParse(token,System.Globalization.NumberStyles.Float,CultureInfo.InvariantCulture,out shown);

  Check(gauge!=null&&gauge.gameObject.activeInHierarchy&&gauge.enabled&&gauge.font!=null&&gauge.font.name.Contains("CutePixel")&&parsed&&Mathf.Abs(shown-value)<.11f,"rendered CutePixel fuel label matches "+value);
  float wanted=Mathf.Lerp(furnace.outOfFuelEmission,1,Mathf.Clamp01(value/(100*furnace.fullBurnFuelFraction)));

 }

  double now=Time.realtimeSinceStartupAsDouble;if(now-began>65){Finish(!driver);return;}
  if(PhotonSession.RunId!=run||PhotonSession.Phase!=RailPhase.Running||RailTrain.Instance==null||RailExpedition.Instance.State.finished)throw new Exception("Run/phase changed; no ending is requested by this fixture");
  Check(Mathf.Abs(RailTrain.Instance.Distance-distance)<.02f&&RailTrain.Instance.Speed<.01f,"train distance and stopped speed retained");ScanAudio(false);
  bool playing=source.isPlaying;if(playing&&!oldPlaying){starts++;sawPlay=true;Log("HORN_START count="+starts+" clip="+source.clip?.name+" duration="+source.clip?.length+" loop="+source.loop);}oldPlaying=playing;
  if(playing)Check(!source.loop,"real horn clip is non-looping");
  float fuel=PhotonSession.CurrentFuel;if(Mathf.Abs(fuel-lastFuel)>.01f){lastFuel=fuel;lastChange=now;Log("FUEL_RECEIVED value="+fuel+" revision="+PhotonSession.Get(RailSessionRules.Revision,0));}
  if(now>=nextLog){nextLog=now+.5;Log("STATE stage="+stage+" fuel="+fuel+" hornPlaying="+playing+" hornHeld="+RailRemoteBridge.Get(horn,"held")+" active="+RailRemoteBridge.Get(horn,"hornActive")+" time="+source.time+" volume="+source.volume+" starts="+starts);}
  if(!driver){if(lastChange>0&&now-lastChange>2.8){SampleFuel(fuel);lastChange=-1;}return;}

  if(stage<targets.Length){
   float target=targets[stage];if(ack<0&&Mathf.Abs(fuel-target)<.01f&&PhotonSession.Get(RailSessionRules.Revision,0)>revision){ack=now;Log("ACK stage="+stage+" target="+target);}
   if(ack>=0&&now-ack>=3){SampleFuel(target);stage++;ack=-1;revision=PhotonSession.Get(RailSessionRules.Revision,0);retry=now+.25;edge=now;return;}
   if(ack<0&&now>=retry){retry=now+1.5;if(stage==1)Fuel100Button();else Log("FUEL_REQUEST target="+target+" accepted="+RailRemoteBridge.Commit(RailSessionRules.Fuel,target));}return;
  }
  if(stage==5&&now-edge>.5){Check(!RailWorld.InputBlocked,"input available before horn");Check(horn.button.Press(),"horn first Press accepted");stage=6;edge=now;starts=0;sawPlay=false;oldPlaying=false;Log("HORN hold starts; 12 seconds");return;}
  if(stage==6){if(now-edge>1)Check((bool)RailRemoteBridge.Get(horn,"held"),"local horn stays held for long press");if(now-edge<12)return;
   Check(sawPlay&&starts==1,"one playback start during twelve-second horn hold");if(source.clip!=null&&source.clip.length<11.5f)Check(!playing,"short one-shot stays finished while held");horn.button.Release();stage=7;edge=now;Log("HORN release, observe two seconds");return;}

  if(stage==8&&now-edge>.35){Check(sawPlay&&playing,"short pulse actually playing before fade release");horn.button.Release();stage=9;edge=now;Log("HORN short pulse release for fade measurement");return;}
  if(stage==9){if(firstFadeStep<0)firstFadeStep=Time.unscaledDeltaTime;if(playing&&source.volume>0&&source.volume<.99f)fadeSample=true;if(now-edge<.5)return;Check(!playing,"short release stops within half a second");
   if(!fadeSample&&firstFadeStep>=.15f){inconclusive=true;Log("BLOCKED fade temporal resolution: first release frame="+firstFadeStep+" exceeds 150ms fade; stop behavior checked separately");}else Check(fadeSample,"intermediate fade volume observed");
   Check(starts==1,"short pulse has one playback start");Check(outCues==1,"one fresh out-of-fuel audio lease");Check(fireCues>=1,"fresh furnace loop lease after refuel");Finish(true);}
 }catch(Exception ex){Log("BLOCKED fixture "+ex);Finish(false);}}

   if(panel!=null){panel.Close();RailRemoteBridge.Set(panel,"tools",oldTools);RailRemoteBridge.Set(panel,"anchor",null);panel.Build();panel.Close();if(oldOpen)panel.Toggle();Cursor.lockState=oldLock;Cursor.visible=oldCursor;}
   if(RailTestTools.CanMutate&&PhotonSession.CurrentFuel<99.99f)Log("CLEANUP fuel100 request accepted="+RailRemoteBridge.Commit(RailSessionRules.Fuel,100f)+"; receipt not verified after abnormal exit");
  }
  Log((failed.Count>0?"FAIL ":inconclusive?"BLOCKED ":complete?(driver?"PASS ":"OBSERVED "):"BLOCKED ")+"DONE driver="+driver+" stage="+stage+" failures="+failed.Count+" settledChecks="+settledChecks+" fuel="+PhotonSession.CurrentFuel+" outCues="+outCues+" fireCues="+fireCues+" fadeObserved="+fadeSample+" duration="+(Time.realtimeSinceStartupAsDouble-began));UnityEngine.Object.Destroy(gameObject);
 }
