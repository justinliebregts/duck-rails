using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using Photon.Pun;
internal class CommandScript:IRunCommand {const string Root="X:/Work/duck-rails/";string role,path;int slot,stage;double entered,finished;Vector3 target,lastHead;bool kin,rescued;string ids;
 void Log(string s)=>File.AppendAllText(path,DateTime.UtcNow.ToString("O")+" "+s+"\n");void Check(bool b,string s)=>Log((b?"PASS ":"FAIL ")+s);
 public void Execute(ExecutionResult r){var callbacks=(EditorApplication.CallbackFunction)RailRemoteBridge.Get(typeof(EditorApplication),"update");foreach(var d in callbacks.GetInvocationList())if(d.Target!=null&&d.Target.GetType().Name=="CommandScript")EditorApplication.update-=(EditorApplication.CallbackFunction)d;role=Unity.Multiplayer.PlayMode.CurrentPlayer.IsMainEditor?"main":Unity.Multiplayer.PlayMode.CurrentPlayer.Tags.First();slot=role=="main"?0:role=="Player2"?1:2;path=Root+"docs/validation/playtest-queue/2026-11-13-regression/finish-corrected-"+role+".txt";entered=EditorApplication.timeSinceStartup;kin=NewPlayer.instance.rb.isKinematic;
 var ds=UnityEngine.Object.FindObjectsByType<PhotonView>(FindObjectsSortMode.None).Where(v=>v.GetComponent<Game.AI.Enemy>()!=null&&v.InstantiationData is object[] d&&d.Length>=3&&d[d.Length-1] is string s&&s.StartsWith("enemy-")&&int.TryParse(s.Substring(6),out int n)&&n>=RailExpedition.Instance.baseEnemyCount).OrderBy(v=>v.ViewID);ids=string.Join(",",ds.Select(v=>v.ViewID+":"+v.SpawnToken));Log("Defenders after handoff/late rejoin="+ids);foreach(int id in File.ReadAllText(Root+"Temp/regression-belowfloor-ids.txt").Split(',').Select(int.Parse))Check(PhotonView.Find(id)==null,"Coal/gold below floor removed view="+id);
 File.WriteAllText(Root+"Temp/regression-finish-ready-"+role,"ready");EditorApplication.update+=Tick;r.Log("Finish trial "+role);}
 void Tick(){if(!Application.isPlaying){EditorApplication.update-=Tick;return;}try{var run=RailExpedition.Instance;var train=RailTrain.Instance;double age=EditorApplication.timeSinceStartup-entered;RailRemoteBridge.Set(UnityEngine.Object.FindFirstObjectByType<PhotonMasterHandoff>(),"focused",true);foreach(var h in UnityEngine.Object.FindObjectsByType<RailHoldControl>(FindObjectsSortMode.None))RailRemoteBridge.Set(h,"focused",true);
 if(age>90){Log("BLOCKED finish timeout stage="+stage);End();return;}
 if(stage==0){if(new[]{"main","Player2","Player3"}.Any(x=>!File.Exists(Root+"Temp/regression-finish-ready-"+x)))return;if(slot<2)RailBridgeHoldFixture.SetHeld(true);stage++;}
 else if(stage==1&&run.State.BridgeProgress(PhotonNetwork.Time)>=1){RailBridgeHoldFixture.SetHeld(false);run.bridgeButton.GetComponent<RailHoldControl>().button.Release();Check(true,"Two simultaneous holders complete retained bridge work");if(slot==0){float distance=RailRoute.Instance.FinaleStart+run.bridgeZ-RailRoute.Instance.finaleRoot.position.z-9;RailRemoteBridge.Commit(RailSessionRules.Motion,new object[]{PhotonNetwork.Time,distance,0f,0f},RailSessionRules.Fuel,80f);}entered=EditorApplication.timeSinceStartup;stage++;}
 else if(stage==2&&age>2){if(Mathf.Abs(train.transform.position.z-(run.bridgeZ-9))>.3f){if(slot==0){float distance=RailRoute.Instance.FinaleStart+run.bridgeZ-RailRoute.Instance.finaleRoot.position.z-9;RailRemoteBridge.Commit(RailSessionRules.Motion,new object[]{PhotonNetwork.Time,distance,0f,0f},RailSessionRules.Fuel,80f);}return;}target=slot==0?new Vector3(2.6f,1,0):slot==1?new Vector3(0,1.5f,6):new Vector3(25,0,0);NewPlayer.instance.rb.isKinematic=true;Place(train);Check(!train.Contains(NewPlayer.instance.headCollider.transform.position)&&train.ContainsPadded(NewPlayer.instance.headCollider.transform.position)==(slot<2),"Spatial boundary tight=false padded="+(slot<2)+" local="+target);entered=EditorApplication.timeSinceStartup;stage++;}
 else if(stage==3){if(!run.State.finished){Place(train);if(age>2&&slot==0)PhotonSession.Instance.SetTrainMoving(true);return;}finished=EditorApplication.timeSinceStartup;var row=run.LocalParticipant;Check(row.escaped==(slot<2),"Actual train crossing escaped="+row.escaped+" reason="+run.State.reason);Log("Board="+JsonUtility.ToJson(run.State));if(slot==2)Check(train.ContainsPadded(NewPlayer.instance.headCollider.transform.position),"Far player teleported to deck on finish");lastHead=NewPlayer.instance.headCollider.transform.position;stage++;}
 else if(stage==4){if(slot<2){var local=train.transform.InverseTransformPoint(NewPlayer.instance.headCollider.transform.position);if(Vector2.Distance(new Vector2(local.x,local.z),new Vector2(target.x,target.z))>1)rescued=true;Place(train);}if(EditorApplication.timeSinceStartup-finished>6){Check(slot==2||!rescued,"Padded player remains at intended boundary without rescue jump for six seconds");Log("DONE");End();}}
 }catch(Exception e){Log("FIXTURE ERROR "+e);End();}}
 void Place(RailTrain train){var p=NewPlayer.instance;p.transform.position+=train.transform.TransformPoint(target)-p.headCollider.transform.position;p.rb.position=p.transform.position;p.ReseedTrackingPose();}
 void End(){RailBridgeHoldFixture.SetHeld(false);if(NewPlayer.instance!=null)NewPlayer.instance.rb.isKinematic=kin;EditorApplication.update-=Tick;}
}
