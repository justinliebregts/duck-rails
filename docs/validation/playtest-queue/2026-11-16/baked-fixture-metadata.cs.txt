using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;
internal class CommandScript:IRunCommand {
[Serializable] class Entry {public string path,owner;public string[] lights;}
[Serializable] class Data {public List<Entry> entries=new List<Entry>();}
public void Execute(ExecutionResult r){var data=new Data();foreach(var guid in AssetDatabase.FindAssets("t:Prefab",new[]{"Assets/DuckRails"})){var path=AssetDatabase.GUIDToAssetPath(guid);var p=AssetDatabase.LoadAssetAtPath<GameObject>(path);foreach(var owner in p.GetComponentsInChildren<RailBakedBlockLighting>(true)){var oid=GlobalObjectId.GetGlobalObjectIdSlow(owner);if(oid.assetGUID.ToString()!=guid)throw new Exception("Nested external owner "+path);var lights=owner.GetComponentsInChildren<Light>(true).Where(l=>l.lightmapBakeType==LightmapBakeType.Baked&&l.GetComponentInParent<RailBakedBlockLighting>(true)==owner).Select(l=>{var id=GlobalObjectId.GetGlobalObjectIdSlow(l);if(id.assetGUID.ToString()!=guid)throw new Exception("External light "+path);return id.targetObjectId.ToString();}).ToArray();data.entries.Add(new Entry{path=path,owner=oid.targetObjectId.ToString(),lights=lights});}}File.WriteAllText("Temp/queue-baked-fixture-metadata.json",JsonUtility.ToJson(data,true));r.Log("Captured "+data.entries.Count+" prefab owners and "+data.entries.Sum(x=>x.lights.Length)+" baked light references without saving assets");}}
