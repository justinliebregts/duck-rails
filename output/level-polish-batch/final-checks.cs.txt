using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using System;using System.IO;using System.Linq;using System.Collections.Generic;using Object=UnityEngine.Object;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult result){RailCanyon.Guard();AssetDatabase.Refresh();EditorSceneManager.OpenScene(RailFinaleFort.BakeScene);var b=Object.FindFirstObjectByType<RailBlock>();var hinge=RailBlock.FindEnvironment(b.transform,"Bridge hinge");var angle=hinge.localRotation;var lines=new List<string>();int count=0;
void Check(bool ok,string name){count++;if(!ok)lines.Add("FAIL "+name);}
try{
Physics.SyncTransforms();foreach(var t in b.GetComponentsInChildren<Transform>().Where(t=>t.name=="Recovery guard post")){var foot=t.position-Vector3.up*t.localScale.y/2;Check(Physics.RaycastAll(foot+Vector3.up*.04f,Vector3.down,.09f,1<<6).Any(h=>h.collider.transform!=t),"post seated "+foot);}
var own=hinge.GetComponentsInChildren<Collider>();var fixedC=b.GetComponentsInChildren<Collider>().Where(c=>!c.transform.IsChildOf(hinge)&&!c.isTrigger).ToArray();
for(int degrees=0;degrees<=80;degrees+=5){hinge.localRotation=Quaternion.Euler(-degrees,0,0);Physics.SyncTransforms();int contacts=0;string first="";foreach(var c in own)foreach(var other in fixedC)if(c.bounds.Intersects(other.bounds)&&Physics.ComputePenetration(c,c.transform.position,c.transform.rotation,other,other.transform.position,other.transform.rotation,out var direction,out float distance)&&distance>.035f){contacts++;first=c.name+" / "+other.name;}Check(contacts==0,"bridge sweep "+degrees+" contacts="+contacts+" "+first);}
hinge.localRotation=Quaternion.identity;Physics.SyncTransforms();for(int z=0;z<310;z++){Check(!Physics.CheckBox(new(0,3.25f,z),new Vector3(2.9f,3.05f,.35f),Quaternion.identity,1<<6,QueryTriggerInteraction.Ignore),"full train corridor "+z);}
var asset=AssetDatabase.LoadAssetAtPath<GameObject>(RailFinaleFort.PrefabPath);var lighting=asset.GetComponent<RailBakedBlockLighting>();Check(lighting.bindings.Length>0&&lighting.bindings.All(x=>x.renderer&&x.color),"saved lightmap references");Check(EditorBuildSettings.scenes.Any(s=>s.enabled&&s.path==lighting.lightingScene),"lighting companion enabled");Check(RailBlock.FindEnvironment(asset.transform,"Bridge hinge").GetComponentsInChildren<Renderer>(true).All(x=>x.enabled),"runtime bridge visible");Check(RailBlock.FindEnvironment(asset.transform,"Far bank").GetComponent<Renderer>().enabled,"runtime far bank visible");Check(asset.GetComponentsInChildren<Light>(true).Count(x=>x.lightmapBakeType==LightmapBakeType.Baked)==18,"18 total baked lights");Check(asset.GetComponent<RailBlock>().graph!=null,"saved navigation reference");Check(asset.GetComponentsInChildren<MeshFilter>(true).All(f=>f.sharedMesh!=null),"no missing meshes");
lines.Insert(0,count+" final checks; "+lines.Count+" failures. EditMode only.");lines.Add("Static bindings: "+lighting.bindings.Length+". Mesh triangles: "+asset.GetComponentsInChildren<MeshFilter>(true).Sum(f=>f.sharedMesh.triangles.Length/3)+". Renderers: "+asset.GetComponentsInChildren<Renderer>(true).Length+".");File.WriteAllLines("output/level-polish-batch/final-checks.txt",lines);result.Log(string.Join("\n",lines));
}finally{hinge.localRotation=angle;Physics.SyncTransforms();EditorSceneManager.OpenScene(RailFinaleFort.BakeScene);}
}}
