using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using Photon.Pun;
internal class CommandScript : IRunCommand {
 const string Root="X:/Work/duck-rails/"; bool main;string role,code;int stage;double began;
 public void Execute(ExecutionResult r){main=Unity.Multiplayer.PlayMode.CurrentPlayer.IsMainEditor;role=main?"main":Unity.Multiplayer.PlayMode.CurrentPlayer.Tags.First();began=EditorApplication.timeSinceStartup;EditorApplication.update+=Tick;r.Log("Bootstrap "+role);}
 void Tick(){if(!Application.isPlaying)return;try{var s=PhotonSession.Instance;if(s==null||!s.WorldReady||s.HasRoomRequest)return;
 if(EditorApplication.timeSinceStartup-began>180){EditorApplication.update-=Tick;return;}
 if(main){if(stage==0&&PhotonSession.Phase==RailPhase.Compound){code=UnityEngine.Random.Range(10000,90000).ToString();s.RequestCreatePrivate(code);stage=1;}
 else if(stage==1&&PhotonSession.Phase==RailPhase.Staging){File.WriteAllText(Root+"Temp/regression-world-room.txt",code);stage=2;}
 else if(stage==2&&PhotonNetwork.CurrentRoom.PlayerCount==3){s.StartPrivateRun();stage=3;}
 else if(stage==3&&PhotonSession.Phase==RailPhase.Running){File.WriteAllText(Root+"docs/validation/playtest-queue/2026-11-13-regression/world-start-postrestart.txt",DateTime.UtcNow.ToString("O")+" ready three distinct="+PhotonNetwork.PlayerList.Select(p=>p.UserId).Distinct().Count()+" route="+string.Join(",",RailRoute.Instance.Selection.blocks));EditorApplication.update-=Tick;}}
 else {if(stage==0&&File.Exists(Root+"Temp/regression-world-room.txt")){code=File.ReadAllText(Root+"Temp/regression-world-room.txt");if(code.Length==5){s.RequestJoinCode(code);stage=1;}}else if(stage==1&&PhotonSession.Phase==RailPhase.Running){SimulatedRigInput.Instance.SetSimulating(true);EditorApplication.update-=Tick;}}
 }catch(Exception e){File.AppendAllText(Root+"docs/validation/playtest-queue/2026-11-13-regression/bootstrap-errors.txt",role+" "+e);EditorApplication.update-=Tick;}}
}
