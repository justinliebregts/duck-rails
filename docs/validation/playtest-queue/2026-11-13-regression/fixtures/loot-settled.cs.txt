using System;using System.IO;using System.Linq;using UnityEngine;using Photon.Pun;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult r){var lines=new System.Collections.Generic.List<string>();foreach(var b in UnityEngine.Object.FindObjectsByType<RailBlock>(FindObjectsSortMode.None)){int total=0,supported=0,still=0;float maxGap=0;foreach(var item in UnityEngine.Object.FindObjectsByType<PhotonGrabbable>(FindObjectsSortMode.None).Where(x=>x.GetComponent<RailCoal>()!=null||x.GetComponent<ItemGoldBar>()!=null)){var pos=b.transform.InverseTransformPoint(item.transform.position);if(pos.z<0||pos.z>b.length)continue;if(item.IsHeldByAnyPlayer||item.IsStashed)continue;var cols=item.GetComponentsInChildren<Collider>().Where(c=>c.enabled&&!c.isTrigger).ToArray();if(cols.Length==0)continue;var bounds=cols[0].bounds;foreach(var c in cols.Skip(1))bounds.Encapsulate(c.bounds);total++;var bottom=new Vector3(bounds.center.x,bounds.min.y,bounds.center.z);bool hit=Physics.Raycast(bottom+Vector3.up*.08f,Vector3.down,out var h,.25f,1<<6,QueryTriggerInteraction.Ignore);float gap=hit?Mathf.Abs(h.point.y-bottom.y):999;maxGap=Mathf.Max(maxGap,gap);if(hit&&gap<.06f)supported++;var rb=item.GetComponent<Rigidbody>();if(rb==null||rb.isKinematic||rb.IsSleeping()||rb.linearVelocity.sqrMagnitude<.0001f)still++;if(!hit||gap>=.06f)lines.Add("FLAG "+b.id+" view="+item.GetComponent<PhotonView>().ViewID+" gap="+gap+" pos="+pos);}
lines.Add((total>0&&supported==total&&still==total?"PASS ":"FLAGGED ")+b.id+" loose selected loot count="+total+" supported="+supported+" settled="+still+" maxGap="+maxGap);}
File.WriteAllLines("X:/Work/duck-rails/docs/validation/playtest-queue/2026-11-13-regression/loot-live-settled.txt",lines);r.Log(string.Join("\n",lines));}}
