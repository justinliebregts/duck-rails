 void Arcade(RailBlock b){foreach(float z in new[]{67.5f,81.5f}){var a=b.transform.TransformPoint(new Vector3(-18.5f,1.65f,z));var e=b.transform.TransformPoint(new Vector3(-19.5f,1.65f,z));bool hit=Physics.SphereCast(a,.08f,(e-a).normalized,out var h,(e-a).magnitude,Mask,QueryTriggerInteraction.Ignore);Check(!hit,"Arcade window 16cm aperture sweep z="+z+" hit="+(hit?Path(h.collider.transform):"none"));drops.Add(new Drop{b=b,p=a,end=e,horizontal=true,label="Arcade window z="+z});}var arc=b.GetComponentsInChildren<Transform>().FirstOrDefault(t=>t.name=="Arcade"&&t.GetComponentsInChildren<Collider>().Length>4);if(arc){Check(arc.GetComponentsInChildren<Collider>().All(c=>c.name!="Clear shop window collision"),"Arcade removed pane colliders absent");Log("INFO Arcade remaining Window frame colliders="+arc.GetComponentsInChildren<Collider>().Count(c=>c.name.StartsWith("Window ")));}else Block("Arcade structural subtree absent");}
