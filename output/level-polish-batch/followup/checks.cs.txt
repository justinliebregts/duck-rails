using UnityEngine;using UnityEditor;using Game.AI;using System;using System.IO;using System.Linq;using System.Collections.Generic;
internal class CommandScript:IRunCommand{public void Execute(ExecutionResult result){
var root=AssetDatabase.LoadAssetAtPath<GameObject>(RailFinaleFort.PrefabPath);var env=root.transform.Find("Env");var p=env.Find("November 11 polish");var report=new List<string>();int count=0,fail=0;
void Check(bool ok,string name){count++;if(!ok){fail++;report.Add("FAIL "+name);}}
var ground=env.Find("Finale approach");var mesh=ground.GetComponent<MeshFilter>().sharedMesh;var old=AssetDatabase.LoadAssetAtPath<UnityEngine.Mesh>("Assets/DuckRails/FinaleFort/Polish/Joined concrete-117.asset");
float Area(UnityEngine.Mesh m){var vs=m.vertices;var ts=m.triangles;float area=0;for(int i=0;i<ts.Length;i+=3){var a=ground.TransformPoint(vs[ts[i]]);var b=ground.TransformPoint(vs[ts[i+1]]);var c=ground.TransformPoint(vs[ts[i+2]]);area+=Vector3.Cross(b-a,c-a).magnitude*.5f;}return area;}
Check(Mathf.Abs(Area(old)-Area(mesh))<.5f,"original clipped ground surface area preserved");Check(ground.GetComponent<MeshRenderer>().scaleInLightmap==2,"ground lightmap scale 2");Check(mesh.uv2.Length==mesh.vertexCount,"ground lightmap UV count");
var vertices=mesh.vertices;var tris=mesh.triangles;float longest=0;for(int i=0;i<tris.Length;i+=3)for(int j=0;j<3;j++)longest=Mathf.Max(longest,ground.TransformVector(vertices[tris[i+j]]-vertices[tris[i+(j+1)%3]]).magnitude);Check(longest<5.658f,"ground maximum triangle edge");
report.Add("Ground triangles "+old.triangles.Length/3+" -> "+tris.Length/3+"; area "+Area(old)+" -> "+Area(mesh)+"; maximum edge "+longest+"m; lightmap scale .25 -> 2 at 2 texels/m.");
foreach(int s in new[]{-1,1}){
var ramp=p.Find("Middle ramp "+s);var m=ramp.GetComponent<MeshFilter>().sharedMesh;Check(m==ramp.GetComponent<MeshCollider>().sharedMesh,"ramp collision matches slab "+s);
foreach(var v in m.vertices){float top=2+(v.z-48)*.25f;Check(Mathf.Abs(v.y-top)<.001f||Mathf.Abs(v.y-(top-.2f))<.001f,"ramp constant vertical thickness "+s);}
var wall=p.Cast<Transform>().First(t=>t.name=="Low approach concrete wall"&&Mathf.Sign(t.position.x)==s);var c=wall.GetComponent<BoxCollider>();var bounds=new Bounds(wall.TransformPoint(c.center),Vector3.Scale(c.size,wall.lossyScale));Check(Mathf.Abs(bounds.min.z)<.001f&&Mathf.Abs(bounds.max.z-80)<.001f,"wall reaches water at Z80 "+s);
foreach(int z in new[]{36,72}){var group=p.Find((s<0?"West":"East")+(z<50?" south":" north")+" floodlight");Check(group&&group.childCount==3,"one complete group "+s+" "+z);var head=group.Find("Sports field floodlight array");var pole=group.Find("Security lamp post");var l=group.GetComponentInChildren<Light>();Check(Vector3.Distance(head.position,new(s*35,10,z))<.001f&&Vector3.Distance(pole.position,new(s*35,7,z))<.001f,"fixture positions preserved");Check(l.type==LightType.Spot&&l.lightmapBakeType==LightmapBakeType.Baked&&l.intensity==95&&l.range==52&&l.spotAngle==112,"fixture settings retained");Check(Vector3.Distance(l.transform.position,head.position+head.forward*.3f)<.001f,"light remains with fixture");}
}
var bank=env.Find("Far bank");var bm=bank.GetComponent<MeshFilter>().sharedMesh;Check(bank.GetComponent<MeshRenderer>().sharedMaterials[1].name=="Ridge rock","dark bank edge material");Check(bank.GetComponent<BoxCollider>()!=null,"bank collision retained");var bv=bm.vertices;var bt=bm.GetTriangles(0);Check(bt.All(i=>bank.TransformPoint(bv[i]).z>=111.499f),"grass starts behind dark 1.5m edge");bt=bm.triangles;for(int i=0;i<bt.Length;i+=3){var a=bank.TransformPoint(bv[bt[i]]);var b=bank.TransformPoint(bv[bt[i+1]]);var c=bank.TransformPoint(bv[bt[i+2]]);if(Mathf.Abs(a.z-110)<.001f&&Mathf.Abs(b.z-110)<.001f&&Mathf.Abs(c.z-110)<.001f)Check(Mathf.Min(a.y,b.y,c.y)>=-.0301f,"no bank face overlapping foundation");}
var block=root.GetComponent<RailBlock>();var graph=CrawlGraph.Load(block.graph);int goal=graph.FindNearestGroundNode(new(0,8,71),2);foreach(var t in root.transform.Find("Enemy Spawns").Cast<Transform>().Where(t=>t.gameObject.activeSelf)){
int start=graph.FindNearestGroundNode(t.position,2);Check(start>=0&&Vector3.Distance(graph.GetPosition(start),t.position)<.8f,"tower nearby ground "+t.name);var seen=new bool[graph.NodeCount];var q=new Queue<int>();q.Enqueue(start);seen[start]=true;while(q.Count>0){int n=q.Dequeue();for(int e=graph.EdgeStart[n];e<graph.EdgeStart[n+1];e++){int next=graph.EdgeTarget[e];if(!seen[next]&&graph.GetSurface(next)==CrawlSurface.Floor){seen[next]=true;q.Enqueue(next);}}}Check(goal>=0&&seen[goal],"floor-only route to button "+t.name);}
report.Insert(0,count+" follow-up checks; "+fail+" failures. EditMode only.");report.Add("Loaded mesh triangles: "+root.GetComponentsInChildren<MeshFilter>(true).Sum(f=>f.sharedMesh.triangles.Length/3)+"; mesh renderers: "+root.GetComponentsInChildren<MeshRenderer>(true).Length+"; graph nodes: "+graph.NodeCount);File.WriteAllLines("output/level-polish-batch/followup/checks.txt",report);result.Log(string.Join("\n",report));
}}
