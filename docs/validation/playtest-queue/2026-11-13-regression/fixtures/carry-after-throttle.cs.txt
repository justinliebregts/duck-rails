internal class CommandScript:IRunCommand{
const string P="X:/Work/duck-rails/docs/validation/playtest-queue/2026-11-13-regression/area-carry-after-throttle.txt";string[] blocks={"container-yard-v1","roadside-motel-v1","convenience-store-v1","farm-silos-v1","skyscraper-works-v1","lakeside-campground-v1"};int index,stage;double at;PhotonGrabbable item;PlayerHand hand;SimulatedRigInput sim;Vector3 start;bool left;
